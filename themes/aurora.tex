\newcommand\deck@gradientbg{%
  \shade[shading=aurora, shading angle=-35] (0,0) rectangle (\paperwidth,\paperheight);
  \fill[aink, fill opacity=0.35, path fading=deckglow] (0,0) circle (9cm);
  \fill[white, fill opacity=0.18, path fading=deckglow] (0.85\paperwidth,0.95\paperheight) circle (8cm);
  \fill[white, fill opacity=0.10] (0.80\paperwidth,0.30\paperheight) circle (1.7cm);
  \fill[white, fill opacity=0.14] (0.92\paperwidth,0.62\paperheight) circle (0.7cm);
  \fill[white, fill opacity=0.20] (0.70\paperwidth,0.82\paperheight) circle (0.32cm);}
\setbeamertemplate{background}{%
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (\paperwidth,\paperheight);
    \ifdeck@divider\deck@gradientbg\else\ifnum\c@framenumber=\@ne\deck@gradientbg\else
      \fill[apink, fill opacity=0.10, path fading=deckglow] (\paperwidth,\paperheight) circle (8cm);
      \fill[aviolet, fill opacity=0.08, path fading=deckglow] (0,0) circle (7cm);
      \shade[shading=aurora] (0,\paperheight-2.6pt) rectangle (\paperwidth,\paperheight);
    \fi\fi
  \end{tikzpicture}}

  \vskip4pt}
\setbeamertemplate{footline}{%
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (\paperwidth,0.6cm);
    \ifnum\c@framenumber>\@ne
      \node[anchor=west, font=\tiny, text=amuted] at (1cm,0.32cm) {\insertshortauthor\hspace{0.5em}\textperiodcentered\hspace{0.5em}\insertshorttitle};
      \node[anchor=east, font=\tiny\bfseries, text=aviolet] at (\paperwidth-1cm,0.32cm) {\insertframenumber\,/\,\inserttotalframenumber};
      \fill[aline] (1cm,0.1cm) rectangle (\paperwidth-1cm,0.135cm);
      \shade[shading=aurora] (1cm,0.1cm) rectangle ({1cm+\insertframenumber/\inserttotalframenumber*(\paperwidth-2cm)},0.135cm);

  \end{tikzpicture}}

\newcommand\deck@tocentry[2]{%
  \leavevmode\parbox[t]{2.4em}{\bfseries\small\color{#1}0\inserttocsectionnumber}%
  \parbox[t]{\dimexpr\linewidth-2.5em}{\raggedright\deckdisplay\bfseries\Large\color{#2}\inserttocsection}\par\vskip9pt}
\setbeamertemplate{section in toc}{\ifdeck@divider\deck@tocentry{white}{white}\else\deck@tocentry{apink}{aink}\fi}
\setbeamertemplate{section in toc shaded}{\deck@tocentry{white!55!apink}{white!55!apink}}
\newcommand\deckdivider{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=south east, inner sep=0pt, text=white, text opacity=0.2, font=\deckdisplay\bfseries\fontsize{170}{170}\selectfont] at ([xshift=-0.5cm,yshift=1.4cm]current page.south east) {0\thesection};
  \end{tikzpicture}%
  \begin{minipage}{0.72\textwidth}
    {\usebeamerfont{kicker}\color{white}\textls[220]{OUTLINE}\par}\vskip12pt
    \tableofcontents[currentsection]
  \end{minipage}}
