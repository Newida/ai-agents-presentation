\newcommand\deckdivider{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=south west, inner sep=0pt, text=white, font=\bfseries\fontsize{160}{160}\selectfont] at ([xshift=0.75cm,yshift=1.35cm]current page.south west) {\textls[-40]{0\thesection}};
  \end{tikzpicture}%
  \hfill\begin{minipage}[c]{0.5\textwidth}
    {\tiny\bfseries\color{white}\textls[200]{OUTLINE}\par}\vskip10pt
    \tableofcontents[currentsection]
  \end{minipage}}
