\setbeamertemplate{background}{%
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (\paperwidth,\paperheight);
    \ifnum\c@framenumber=\@ne
      \fill[nviolet, fill opacity=0.55, path fading=deckglow] (0.8\paperwidth,0.78\paperheight) circle (9cm);
      \fill[ncyan, fill opacity=0.25, path fading=deckglow] (0.05\paperwidth,0.1\paperheight) circle (7cm);
      \begin{scope}
        \clip (0,0) rectangle (\paperwidth,2.3cm);
        \foreach \k in {1,...,9}{\draw[ncyan, line width=0.35pt, opacity=0.5] (0,{2.3cm*(\k/9)^2}) -- (\paperwidth,{2.3cm*(\k/9)^2});}
        \foreach \x in {-14,...,14}{\draw[ncyan, line width=0.35pt, opacity=0.5] ({0.5*\paperwidth+\x*1.2cm},0) -- ({0.5*\paperwidth+\x*0.2cm},2.3cm);}
      \end{scope}
      \fill[nbg, path fading=north] (0,-0.05cm) rectangle (\paperwidth,2.35cm);
      \shade[shading=neon] (0,2.3cm) rectangle (\paperwidth,2.34cm);
    \else\ifdeck@divider
      \fill[nviolet, fill opacity=0.5, path fading=deckglow] (0.2\paperwidth,0.5\paperheight) circle (8cm);
      \fill[npink, fill opacity=0.18, path fading=deckglow] (\paperwidth,0) circle (7cm);
    \else
      \fill[nviolet, fill opacity=0.32, path fading=deckglow] (\paperwidth,\paperheight) circle (9cm);
      \fill[ncyan, fill opacity=0.14, path fading=deckglow] (0,0) circle (8cm);
    \fi\fi
  \end{tikzpicture}}

  \end{minipage}}
