% Shared typography for the restrained Paper studies. No cover ornaments,
% oversized section numbers, gradients, small caps, or repeated running titles.
% Each study supplies its palette, font setup, cover, and block treatment.
\input{themes/common}
\makeatletter
\paperfonts
\ifnum\paperbody=0
  \usefonttheme{serif}
\else
  \renewcommand\familydefault{\sfdefault}
\fi
\setbeamersize{text margin left=0.85cm,text margin right=0.85cm}
\arrayrulecolor{edrule}
\definecolor{warn}{HTML}{A43A30}
\definecolor{good}{HTML}{43634B}
\definecolor{chartmark}{HTML}{FF9B7B}
\setbeamercolor{background canvas}{bg=edbg}
\setbeamercolor{normal text}{fg=edink,bg=edbg}
\setbeamercolor{structure}{fg=edaccent}
\setbeamercolor{frametitle}{fg=edink}
\setbeamercolor{block title}{fg=edink}
\setbeamercolor{block body}{fg=edink}
\setbeamercolor{itemize item}{fg=edaccent}
\setbeamercolor{enumerate item}{fg=edaccent}
\setbeamercolor{itemize subitem}{fg=edmuted}
\setbeamerfont{frametitle}{family=\rmfamily,series=\mdseries,size=\LARGE}
\setbeamerfont{block title}{family=\sffamily,series=\bfseries,size=\small}
% Inherit the surrounding size: enlarging an already-dense footnotesize slide's
% block would force Beamer to shrink all of its text unnecessarily.
\setbeamerfont{block body}{size={}}
\setbeamertemplate{headline}{}
\setbeamertemplate{frametitle}{%
  \vskip0.25cm
  {\raggedright\usebeamerfont{frametitle}\color{edink}\insertframetitle\par}%
  \vskip5pt\nointerlineskip
  \hbox{\color{edrule}\rule{\textwidth}{0.4pt}}\vskip7pt}
\setbeamertemplate{footline}{%
  \ifnum\c@framenumber>\@ne
    \hbox to\paperwidth{\hfil
      {\sffamily\fontsize{8}{9}\selectfont\color{edmuted}\insertframenumber}%
      \hspace{0.85cm}}\vskip8pt
  \fi}
\setbeamertemplate{itemize item}{\raisebox{0.1ex}{\scriptsize\textbullet}}
\setbeamertemplate{itemize subitem}{\color{edmuted}\textendash}
\setbeamertemplate{enumerate item}{\color{edaccent}\insertenumlabel.}
\renewcommand\deckplus{{\color{good}\bfseries+}}
\renewcommand\deckminus{{\color{warn}\bfseries\textendash}}
\tcbset{deck block/.style={enhanced,sharp corners,boxrule=0pt,
  colback=edpanel,colframe=edrule,coltext=edink,
  left=8pt,right=8pt,top=6pt,bottom=6pt,before skip=4pt,after skip=3pt}}

% Small, upright section numbers support navigation instead of decorating it.
\newcommand\paperentry[2]{%
  \leavevmode\parbox[t]{1.5em}{\sffamily\normalsize\color{#1}\inserttocsectionnumber.}%
  \parbox[t]{\dimexpr\linewidth-1.6em}{\raggedright\large\color{#2}\inserttocsection}%
  \par\vskip13pt}
\setbeamertemplate{section in toc}{\paperentry{edaccent}{edink}}
\setbeamertemplate{section in toc shaded}{\paperentry{edmuted}{edmuted}}
\newcommand\deckdivider{%
  \begin{minipage}{0.94\textwidth}
    {\usebeamerfont{frametitle}\color{edink}Outline\par}\vskip18pt
    \tableofcontents[currentsection]
  \end{minipage}}

\newcommand\paperheading[2]{%
  {\raggedright\rmfamily\fontsize{#1}{#2}\selectfont\color{edink}\inserttitle\par}}

\newcommand\papermetadata{%
  {\sffamily\small\color{edink}\insertauthor\par}\vskip3pt
  {\sffamily\footnotesize\color{edmuted}\insertdate\par}}
