\setbeamertemplate{frametitle}{%
  \vskip0.4cm
  \deckifsection{{\footnotesize\scshape\color{clay}\textls[40]{\@Roman\c@section\enspace{\upshape\textperiodcentered}\enspace\decksectionlc}\par}\vskip1pt}{}%
  {\usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}\insertframetitle\par}%
  \vskip4pt\nointerlineskip\hbox{\color{paperrule}\rule{\textwidth}{0.5pt}}\vskip6pt}
\setbeamertemplate{footline}{%
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (\paperwidth,0.62cm);
    \ifnum\c@framenumber>\@ne
      \node[anchor=west, font=\scriptsize\scshape, text=papermuted] at (1cm,0.32cm) {\insertshorttitle\enspace{\upshape\textperiodcentered}\enspace\insertshortauthor};
      \node[anchor=east, font=\footnotesize\itshape, text=clay] at (\paperwidth-1cm,0.32cm) {\insertframenumber};
    \fi
  \end{tikzpicture}}

    \fill[clay] ([xshift=-2.8cm,yshift=1.9cm]current page.south east) circle (1.4cm);

  \end{tikzpicture}%
  {\footnotesize\scshape\color{clay}\textls[40]{\insertdate}\par}\vskip10pt

  {\color{clay}\rule{2.5cm}{0.8pt}\par}\vskip10pt
  {\small\scshape\color{paperink}\insertauthor\par}}

\newcommand\deck@tocentry[2]{%
  \leavevmode\parbox[t]{2em}{\itshape\color{#1}\@Roman{\inserttocsectionnumber}.}%
  \parbox[t]{\dimexpr\linewidth-2.1em}{\raggedright\Large\color{#2}\inserttocsection}\par\vskip9pt}

\newcommand\deckdivider{%
  \begin{minipage}[c]{0.3\textwidth}\raggedleft
    {\fontsize{120}{120}\selectfont\itshape\color{clay}\@Roman\c@section}
  \end{minipage}\hspace{0.04\textwidth}%
  {\color{paperrule}\vrule width 0.6pt height 2.2cm depth 1.2cm}\hspace{0.04\textwidth}%
  \begin{minipage}[c]{0.58\textwidth}
    {\footnotesize\scshape\color{clay}\textls[60]{outline}\par}\vskip10pt
    \tableofcontents[currentsection]
  \end{minipage}}
