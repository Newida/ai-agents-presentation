\begin{frame}[shrink=0]{Field Report: Automating Exercises}

    \setlength{\itemsep}{0.55em}
    \item<1-> \textbf{The problem.} PhD student, tutoring an exercise class every two weeks:
      \begin{itemize}\footnotesize

        \item rough estimate: \textbf{2\,h + 6\,h = 8\,h every two weeks}

\begin{frame}[shrink=0]{Automating at Bigger Scale}
  \begin{itemize}
    \item<1-> \textbf{So far:} each of us automates our own tasks
    \item<2-> \textbf{Share it:} one person builds it, everyone reuses and adapts it to their needs \\
    {\color{warn} We need a place our agents can look for shared Lamarr automatizations}
    \item<3-> \textbf{Institute level:} tasks that never pay off for one person start to pay off
  \end{itemize}
  \vspace{-3pt}
  \uncover<4->{%
  \begin{block}{Example: travel expense reports}
    \footnotesize Collect receipts, fill in the university form, look up per-diem rules, fix what admin sends back: say \textbf{3 trips $\times$ 3\,h per person per year}. An agent that reads the receipts and files the form correctly needs \textbf{$\sim$200\,h} (rules, forms, access to the admin system)\par\smallskip
    \centering
    \begin{tabular}{lccc}
      & how often $\times$ time saved & time to build & \\ \hline
      \uncover<5->{one person} & \uncover<5->{$3 \times 3$\,h $= 9$\,h/year} & \uncover<5->{200\,h} & \uncover<5->{\textcolor{warn}{\textbf{never pays}}} \\
      \uncover<6->{Lamarr, $\sim$300 researchers} & \uncover<6->{$300 \times 9$\,h $= 2\,700$\,h/year} & \uncover<6->{200\,h} & \uncover<6->{\textcolor{good}{\textbf{pays $>$13$\times$}}} \\
    \end{tabular}
  \end{block}}
\end{frame}

\begin{frame}[shrink=0]{Decision 1: Subscription or API?}
  \small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \uncover<1->{\textbf{Subscription} (20--200\,\euro{}/month)
      \begin{itemize}
        \item[\deckplus] flat price, no surprise bills
        \item[\deckplus] very cheap for heavy interactive use
        \item[\deckminus] usage caps (5-hour and weekly)
        \item[\deckminus] one person, one login; not ideal for pipelines
      \end{itemize}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \uncover<2->{\textbf{API} (pay per token)
      \begin{itemize}
        \item[\deckplus] no caps, run many agents in parallel
        \item[\deckplus] scripts, batch jobs, shared tools
        \item[\deckminus] heavy daily use gets expensive fast
        \item[\deckminus] needs a budget and someone watching it
      \end{itemize}}
    \end{column}
  \end{columns}
